import logging
import time
from urllib.parse import urlparse

import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

log = logging.getLogger("radar")

# Seconds between requests to the same host. Conservative on purpose:
# Steam appdetails tolerates ~200 calls / 5 min, Apple's iTunes endpoints ~20/min.
HOST_INTERVALS = {
    "store.steampowered.com": 1.6,
    "steamspy.com": 1.0,
    "itunes.apple.com": 3.1,
    "api.igdb.com": 0.3,
    "itch.io": 1.5,
    "www.appbrain.com": 6.0,   # AppBrain answers 429 after ~20 fast requests
    "play.google.com": 0.7,
}


# Lighter endpoints on the same hosts can go faster than the host default.
PATH_INTERVALS = [
    ("store.steampowered.com", "/appreviews/", 0.4),   # review summaries: far more lenient than appdetails
    ("itunes.apple.com", "/rss/", 1.0),                # chart feeds: CDN-served, unlike the Lookup API
]


class Http:
    def __init__(self, user_agent: str, default_interval: float = 1.0, timeout: int = 30):
        s = requests.Session()
        retry = Retry(total=5, backoff_factor=2, status_forcelist=(429, 500, 502, 503, 504),
                      allowed_methods=frozenset(["GET", "POST"]), respect_retry_after_header=True)
        s.mount("https://", HTTPAdapter(max_retries=retry))
        s.mount("http://", HTTPAdapter(max_retries=retry))
        s.headers["User-Agent"] = user_agent
        self.s, self.timeout, self.default_interval = s, timeout, default_interval
        self._last = {}

    def _throttle(self, url):
        u = urlparse(url)
        key, gap = u.netloc, HOST_INTERVALS.get(u.netloc, self.default_interval)
        for host, fragment, interval in PATH_INTERVALS:
            if u.netloc == host and fragment in u.path:
                key, gap = host + fragment, interval
                break
        wait = gap - (time.monotonic() - self._last.get(key, 0.0))
        if wait > 0:
            time.sleep(wait)
        self._last[key] = time.monotonic()

    def get(self, url, **kw):
        self._throttle(url)
        r = self.s.get(url, timeout=self.timeout, **kw)
        r.raise_for_status()
        return r

    def post(self, url, **kw):
        self._throttle(url)
        r = self.s.post(url, timeout=self.timeout, **kw)
        r.raise_for_status()
        return r

"""Google Play. Play has no public 'new releases' listing, so new games are discovered
through AppBrain, which tracks every Play app and publishes, for each of the 17 game
categories, the most-installed and trending games released in the past 30 days. Each
game's release date, installs, ratings and price then come from Google Play itself
(google-play-scraper). Optional genre-seeded Play searches can add more, but in practice
they mostly return established games."""
import re

from ..dates import parse_date
from ..links import add_link
from ..models import Game
from .base import Source, SourceUnavailable, log

APPBRAIN = "https://www.appbrain.com/apps/{sort}/{cat}/new"
APPBRAIN_CATS = ["games", "action", "adventure", "arcade", "board-games", "card-games", "casino", "casual",
                 "educational", "music-games", "puzzle", "racing", "role-playing", "simulation",
                 "sports-games", "strategy", "trivia", "word-games"]
APPBRAIN_SORTS = ["popular", "trending"]
_AB_TAG = re.compile(r'<a\b[^>]*\bclass="[^"]*\bbrowse-app\b[^"]*"[^>]*>', re.S)
_AB_HREF = re.compile(r'href="/app/[^"/]*/([\w.]+)"')


def parse_appbrain(html):
    """Package ids from an AppBrain list page, in list order (main list only, not sidebars)."""
    out = []
    for tag in _AB_TAG.findall(html or ""):
        m = _AB_HREF.search(tag)
        if m and m.group(1) not in out:
            out.append(m.group(1))
    return out


DEFAULT_QUERIES = [
    "new game", "new puzzle game", "merge game", "match 3", "idle game", "idle rpg",
    "runner game", "arcade game", "casual game", "hyper casual", "strategy game",
    "tower defense", "rpg game", "action game", "roguelike", "survivor game", "simulation game",
    "tycoon", "racing game", "sports game", "card game", "board game", "word game",
    "trivia game", "casino game", "shooter game", "adventure game", "music game", "kids game",
]


def parse_app(d, cc):
    if not str(d.get("genreId") or "").startswith("GAME"):
        return None
    rd = parse_date(d.get("released"))
    cats = [c.get("name") for c in d.get("categories") or [] if c.get("name")] or [d.get("genre")]
    score = d.get("score")
    return Game(
        source="googleplay", source_id=d["appId"], title=d.get("title", ""),
        release_date=rd.isoformat() if rd else None, platforms=["Android"],
        raw_genres=[c for c in cats if c], developer=d.get("developer"),
        price=d.get("price"), currency=d.get("currency"), is_free=d.get("free"),
        has_iap=d.get("offersIAP"), rating=round(score * 20, 1) if score else None,
        rating_count=d.get("ratings"), installs=d.get("realInstalls") or d.get("minInstalls"),
        description=d.get("description") or "", url=d.get("url") or "", icon_url=d.get("icon") or "",
        content_rating=d.get("contentRating"),
        links=add_link(add_link({}, d.get("url") or f"https://play.google.com/store/apps/details?id={d['appId']}", "Google Play"),
                       d.get("developerWebsite"), "Website"),
        extra={"storefront": cc, "genre_id": d.get("genreId"), "installs_label": d.get("installs"),
               "ads": d.get("adSupported"), "last_update": d.get("updated")},
    )


class GooglePlaySource(Source):
    name = "googleplay"

    def _lib(self):
        try:
            from google_play_scraper import app, search
            return app, search
        except ImportError as e:
            raise SourceUnavailable("pip install google-play-scraper to enable Google Play") from e

    def discover_appbrain(self, day=None):
        """AppBrain refuses (HTTP 429) after ~18-20 pages per visit, so by default each run reads
        one list type (most installed / trending) for all 18 categories, alternating by day.
        Both lists cover the past 30 days, so alternating loses nothing."""
        found = {}
        sorts = self.cfg.get("appbrain_sorts", APPBRAIN_SORTS)
        if self.cfg.get("appbrain_alternate", True) and day is not None and len(sorts) > 1:
            sorts = [sorts[day.toordinal() % len(sorts)]]
        for sort in sorts:
            for cat in self.cfg.get("appbrain_categories", APPBRAIN_CATS):
                try:
                    ids = parse_appbrain(self.http.get(APPBRAIN.format(sort=sort, cat=cat)).text)
                except Exception as e:
                    log.warning("appbrain %s/%s failed: %s", sort, cat, e)
                    continue
                for rank, aid in enumerate(ids, 1):
                    found.setdefault(aid, []).append(f"{sort}/{cat}#{rank}")
        log.info("googleplay: AppBrain %s lists: %d games released in the past 30 days", "+".join(sorts), len(found))
        return found

    def fetch(self, start, end):
        app, search = self._lib()
        lang = self.cfg.get("lang", "en")
        cc = self.cfg.get("countries", ["us"])[0]
        found = self.discover_appbrain(end) if self.cfg.get("appbrain", True) else {}
        if self.cfg.get("search", False):
            n = int(self.cfg.get("hits_per_query", 30))
            for q in self.cfg.get("queries", DEFAULT_QUERIES):
                try:
                    for h in search(q, lang=lang, country=cc, n_hits=n):
                        if h.get("appId"):
                            found.setdefault(h["appId"], [])
                except Exception as e:
                    log.debug("play search %r: %s", q, e)
        todo = [a for a in found if a not in self.skip]
        log.info("googleplay: %d not seen before, fetching details from Google Play", len(todo))
        for aid in todo:
            try:
                g = parse_app(app(aid, lang=lang, country=cc), cc)
            except Exception as e:
                log.debug("play app %s: %s", aid, e)
                continue
            if g:
                if found[aid]:
                    g.extra["appbrain"] = found[aid]
                yield g

    def refresh(self, ids):
        app, _ = self._lib()
        cc = self.cfg.get("countries", ["us"])[0]
        for aid in ids:
            try:
                g = parse_app(app(aid, lang=self.cfg.get("lang", "en"), country=cc), cc)
                if g:
                    yield g
            except Exception as e:
                log.debug("play refresh %s: %s", aid, e)

from datetime import date

from radar.db import DB
from radar.links import label_for, ordered
from radar.models import Game
from radar.report import write_daily
from radar.sources.appstore import parse_lookup
from radar.sources.igdb import parse_game
from radar.sources.steam import parse_details
from radar.taxonomy import classify


def test_labels_and_order():
    assert label_for("https://store.steampowered.com/app/1/")[0] == "Steam"
    assert label_for("https://maker.itch.io/game")[0] == "itch.io"
    assert label_for("https://www.nintendo.com/us/store/products/x")[0] == "Nintendo eShop"
    assert label_for("https://mygame.com")[0] == "Website"
    links = {"YouTube": "https://youtube.com/x", "Website": "https://g.com", "IGDB": "https://igdb.com/g", "Steam": "https://store.steampowered.com/app/1/"}
    assert list(ordered(links)) == ["Steam", "Website", "YouTube", "IGDB"]


def test_sources_collect_links():
    g = parse_game({"id": 1, "name": "X", "url": "https://www.igdb.com/games/x",
                    "websites": [{"url": "https://store.steampowered.com/app/9/"}, {"url": "https://store.playstation.com/x"},
                                 {"url": "https://x.com/game"}, {"url": "https://xgame.com"}]})
    assert {"Steam", "PlayStation Store", "Website", "X", "IGDB"} <= set(g.links)
    s = parse_details("9", {"type": "game", "name": "X", "website": "https://xgame.com", "release_date": {"date": "8 Oct, 2026"}})
    assert s.links == {"Steam": "https://store.steampowered.com/app/9/", "Website": "https://xgame.com"}
    a = parse_lookup({"kind": "software", "trackId": 5, "genres": ["Games"], "trackViewUrl": "https://apps.apple.com/app/id5",
                      "sellerUrl": "https://studio.com"}, "us")
    assert a.links == {"App Store": "https://apps.apple.com/app/id5", "Website": "https://studio.com"}


def test_merged_game_shows_every_store(tmp_path):
    db = DB(tmp_path / "r.db")
    day = "2026-10-08"
    for g in (Game(source="steam", source_id="9", title="Star Grid Arena", release_date=day, raw_genres=["Strategy"],
                   links={"Steam": "https://store.steampowered.com/app/9/", "Website": "https://sga.com"}),
              Game(source="googleplay", source_id="com.sga", title="Star Grid Arena", release_date=day, raw_genres=["Strategy"],
                   links={"Google Play": "https://play.google.com/store/apps/details?id=com.sga"})):
        db.upsert(classify(g), day)
    out, rows = write_daily(db, date(2026, 10, 8), tmp_path / "rep")
    assert [l["label"] for l in rows[0]["links"]] == ["Steam", "Google Play", "Website"]
    md = (out / "summary.md").read_text()
    assert "[Star Grid Arena](https://store.steampowered.com/app/9/)" in md and "[Google Play](https://play.google.com" in md
    csv_text = (out / "games.csv").read_text()
    assert "steam_url" in csv_text and "https://sga.com" in csv_text
    assert 'href="https://store.steampowered.com/app/9/"' not in (out / "report.html").read_text()  # data is JSON, rendered by JS
    assert "https://store.steampowered.com/app/9/" in (out / "report.html").read_text()


def test_steam_hover_tags_and_retag(tmp_path, monkeypatch):
    from radar.sources import steam
    html = '<div class="hover_tag_row"><div class="app_tag">Roguelite</div><div class="app_tag">Deckbuilding</div><div class="app_tag">Rock &amp; Roll</div></div>'
    assert steam.parse_hover_tags(html) == ["Roguelite", "Deckbuilding", "Rock & Roll"]
    from radar import pipeline
    db = DB(tmp_path / "r.db")
    g = classify(Game(source="steam", source_id="7", title="Cards of Doom", release_date="2026-10-01", raw_genres=["Action", "Indie"]))
    assert g.genre == "Action"
    db.upsert(g, "2026-10-02"); db.commit()
    monkeypatch.setattr(pipeline, "fetch_tags", lambda http, appid: ["Roguelite", "Deckbuilding"])
    assert pipeline.retag_steam({}, db) == ["2026-10-02"]
    row = db.c.execute("SELECT genre FROM games WHERE key='steam:7'").fetchone()
    assert row["genre"] == "Roguelike"
    assert pipeline.retag_steam({}, db) == []   # nothing left to do


def test_appbrain_parse_and_itch_flag():
    from radar.sources.googleplay import parse_appbrain
    html = ('<div class="sidebar"><a href="/app/x/com.side.bar">Side</a></div>'
            '<a class="browse-app safelink hover-shadow d-block" href="/app/overpeak-action/com.rusergames.overpeak"><h3>OverPeak</h3></a>'
            '<a href="/app/fruit-wins/com.fruitwins" class="browse-app safelink">Fruit</a>'
            '<a class="browse-app" href="/app/overpeak-action/com.rusergames.overpeak">dup</a>')
    assert parse_appbrain(html) == ["com.rusergames.overpeak", "com.fruitwins"]
    from radar.dashboard import aggregate, itch_only
    a = {"release_date": "2026-10-08", "genre": "Puzzle", "subgenres": [], "families": ["Web"], "links": [{"label": "itch.io", "url": "u"}]}
    b = {"release_date": "2026-10-08", "genre": "Puzzle", "subgenres": [], "families": ["Android"], "links": [{"label": "Google Play", "url": "u"}]}
    assert itch_only(a) and not itch_only(b)
    agg = aggregate([a, b])["2026-10-08"]
    assert agg["all"]["n"] == 2 and agg["core"]["n"] == 1


def test_appbrain_alternates_and_path_throttle(monkeypatch):
    from datetime import date
    from radar.sources.googleplay import GooglePlaySource
    seen = []

    class FakeHttp:
        def get(self, url, **kw):
            seen.append(url)
            class R: text = '<a class="browse-app" href="/app/x/com.a.b">x</a>'
            return R()
    src = GooglePlaySource(FakeHttp(), {})
    src.discover_appbrain(date(2026, 10, 12))
    a = {u.split("/")[4] for u in seen}; seen.clear()
    src.discover_appbrain(date(2026, 10, 13))
    b = {u.split("/")[4] for u in seen}
    assert len(a) == 1 and len(b) == 1 and a != b      # one list type per day, alternating
    from radar import http as h
    t = h.Http("x"); sleeps = []
    monkeypatch.setattr(h.time, "sleep", lambda s: sleeps.append(s))
    t._throttle("https://store.steampowered.com/appreviews/1"); t._throttle("https://store.steampowered.com/appreviews/2")
    assert sleeps and max(sleeps) <= 0.4
