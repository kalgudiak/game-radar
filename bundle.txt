"""Google Play. Play has no public 'new releases' listing, so new games are discovered
through AppBrain, which tracks every Play app and publishes, for each of the 17 game
categories, the most-installed and trending games released in the past 30 days. Each
game's release date, installs, ratings and price then come from Google Play itself
(google-play-scraper). Optional genre-seeded Play searches can add more, but in practice
they mostly return established games."""
import re

from ..dates import parse_date
from ..links import add_link
from ..models import Game
from .base import Source, SourceUnavailable, log

APPBRAIN = "https://www.appbrain.com/apps/{sort}/{cat}/new"
APPBRAIN_CATS = ["games", "action", "adventure", "arcade", "board-games", "card-games", "casino", "casual",
                 "educational", "music-games", "puzzle", "racing", "role-playing", "simulation",
                 "sports-games", "strategy", "trivia", "word-games"]
APPBRAIN_SORTS = ["popular", "trending"]
_AB_TAG = re.compile(r'<a\b[^>]*\bclass="[^"]*\bbrowse-app\b[^"]*"[^>]*>', re.S)
_AB_HREF = re.compile(r'href="/app/[^"/]*/([\w.]+)"')


def parse_appbrain(html):
    """Package ids from an AppBrain list page, in list order (main list only, not sidebars)."""
    out = []
    for tag in _AB_TAG.findall(html or ""):
        m = _AB_HREF.search(tag)
        if m and m.group(1) not in out:
            out.append(m.group(1))
    return out


DEFAULT_QUERIES = [
    "new game", "new puzzle game", "merge game", "match 3", "idle game", "idle rpg",
    "runner game", "arcade game", "casual game", "hyper casual", "strategy game",
    "tower defense", "rpg game", "action game", "roguelike", "survivor game", "simulation game",
    "tycoon", "racing game", "sports game", "card game", "board game", "word game",
    "trivia game", "casino game", "shooter game", "adventure game", "music game", "kids game",
]


def parse_app(d, cc):
    if not str(d.get("genreId") or "").startswith("GAME"):
        return None
    rd = parse_date(d.get("released"))
    cats = [c.get("name") for c in d.get("categories") or [] if c.get("name")] or [d.get("genre")]
    score = d.get("score")
    return Game(
        source="googleplay", source_id=d["appId"], title=d.get("title", ""),
        release_date=rd.isoformat() if rd else None, platforms=["Android"],
        raw_genres=[c for c in cats if c], developer=d.get("developer"),
        price=d.get("price"), currency=d.get("currency"), is_free=d.get("free"),
        has_iap=d.get("offersIAP"), rating=round(score * 20, 1) if score else None,
        rating_count=d.get("ratings"), installs=d.get("realInstalls") or d.get("minInstalls"),
        description=d.get("description") or "", url=d.get("url") or "", icon_url=d.get("icon") or "",
        content_rating=d.get("contentRating"),
        links=add_link(add_link({}, d.get("url") or f"https://play.google.com/store/apps/details?id={d['appId']}", "Google Play"),
                       d.get("developerWebsite"), "Website"),
        extra={"storefront": cc, "genre_id": d.get("genreId"), "installs_label": d.get("installs"),
               "ads": d.get("adSupported"), "last_update": d.get("updated")},
    )


class GooglePlaySource(Source):
    name = "googleplay"

    def _lib(self):
        try:
            from google_play_scraper import app, search
            return app, search
        except ImportError as e:
            raise SourceUnavailable("pip install google-play-scraper to enable Google Play") from e

    def discover_appbrain(self):
        found = {}
        for sort in self.cfg.get("appbrain_sorts", APPBRAIN_SORTS):
            for cat in self.cfg.get("appbrain_categories", APPBRAIN_CATS):
                try:
                    ids = parse_appbrain(self.http.get(APPBRAIN.format(sort=sort, cat=cat)).text)
                except Exception as e:
                    log.warning("appbrain %s/%s failed: %s", sort, cat, e)
                    continue
                for rank, aid in enumerate(ids, 1):
                    found.setdefault(aid, []).append(f"{sort}/{cat}#{rank}")
        log.info("googleplay: AppBrain lists %d games released in the past 30 days", len(found))
        return found

    def fetch(self, start, end):
        app, search = self._lib()
        lang = self.cfg.get("lang", "en")
        cc = self.cfg.get("countries", ["us"])[0]
        found = self.discover_appbrain() if self.cfg.get("appbrain", True) else {}
        if self.cfg.get("search", False):
            n = int(self.cfg.get("hits_per_query", 30))
            for q in self.cfg.get("queries", DEFAULT_QUERIES):
                try:
                    for h in search(q, lang=lang, country=cc, n_hits=n):
                        if h.get("appId"):
                            found.setdefault(h["appId"], [])
                except Exception as e:
                    log.debug("play search %r: %s", q, e)
        todo = [a for a in found if a not in self.skip]
        log.info("googleplay: %d not seen before, fetching details from Google Play", len(todo))
        for aid in todo:
            try:
                g = parse_app(app(aid, lang=lang, country=cc), cc)
            except Exception as e:
                log.debug("play app %s: %s", aid, e)
                continue
            if g:
                if found[aid]:
                    g.extra["appbrain"] = found[aid]
                yield g

    def refresh(self, ids):
        app, _ = self._lib()
        cc = self.cfg.get("countries", ["us"])[0]
        for aid in ids:
            try:
                g = parse_app(app(aid, lang=self.cfg.get("lang", "en"), country=cc), cc)
                if g:
                    yield g
            except Exception as e:
                log.debug("play refresh %s: %s", aid, e)

import logging
import time
from urllib.parse import urlparse

import requests
from requests.adapters import HTTPAdapter
from urllib3.util.retry import Retry

log = logging.getLogger("radar")

# Seconds between requests to the same host. Conservative on purpose:
# Steam appdetails tolerates ~200 calls / 5 min, Apple's iTunes endpoints ~20/min.
HOST_INTERVALS = {
    "store.steampowered.com": 1.6,
    "steamspy.com": 1.0,
    "itunes.apple.com": 3.1,
    "api.igdb.com": 0.3,
    "itch.io": 1.5,
    "www.appbrain.com": 2.0,
    "play.google.com": 0.7,
}


class Http:
    def __init__(self, user_agent: str, default_interval: float = 1.0, timeout: int = 30):
        s = requests.Session()
        retry = Retry(total=5, backoff_factor=2, status_forcelist=(429, 500, 502, 503, 504),
                      allowed_methods=frozenset(["GET", "POST"]), respect_retry_after_header=True)
        s.mount("https://", HTTPAdapter(max_retries=retry))
        s.mount("http://", HTTPAdapter(max_retries=retry))
        s.headers["User-Agent"] = user_agent
        self.s, self.timeout, self.default_interval = s, timeout, default_interval
        self._last = {}

    def _throttle(self, url):
        host = urlparse(url).netloc
        gap = HOST_INTERVALS.get(host, self.default_interval)
        wait = gap - (time.monotonic() - self._last.get(host, 0.0))
        if wait > 0:
            time.sleep(wait)
        self._last[host] = time.monotonic()

    def get(self, url, **kw):
        self._throttle(url)
        r = self.s.get(url, timeout=self.timeout, **kw)
        r.raise_for_status()
        return r

    def post(self, url, **kw):
        self._throttle(url)
        r = self.s.post(url, timeout=self.timeout, **kw)
        r.raise_for_status()
        return r

def build_sources(cfg, http, only=None):
    out = []
    for name, cls in REGISTRY.items():
        scfg = (cfg.get("sources") or {}).get(name) or {}
        if only and name not in only:
            continue
        if not only and not scfg.get("enabled", True):
            continue
        out.append(cls(http, scfg))
    return out


def retag_steam(cfg, db, limit=600):
    """Catch-up: fetch community tags for stored Steam games that have none, reclassify them,
    and return the report days whose genres changed so those reports can be rebuilt."""
    http = Http(cfg.get("user_agent", "GameRadar/1.0"))
    rows = db.c.execute("SELECT key, source_id, data, first_seen FROM games WHERE source='steam' "
                        "ORDER BY release_date DESC").fetchall()
    todo = [r for r in rows if not json.loads(r["data"]).get("tags")][:limit]
    changed, days = 0, set()
    for r in todo:
        g = Game.from_dict(json.loads(r["data"]))
        tags = fetch_tags(http, r["source_id"])
        if not tags:
            continue
        old = g.genre
        g.tags = tags
        classify(g)
        db.c.execute("UPDATE games SET data=?, genre=? WHERE key=?", (json.dumps(g.to_dict()), g.genre, r["key"]))
        changed += 1
        days.add(r["first_seen"])
        if changed % 100 == 0:
            db.commit()
    db.commit()
    if todo:
        log.info("steam retag: %d of %d untagged games now tagged", changed, len(todo))
    return sorted(days)


def collect(cfg, db, run_day: date, only=None, sources=None):
    """Discover releases in [run_day - lookback, run_day], classify, store, then
    snapshot metrics for everything released in the last `track_days`."""
    start = run_day - timedelta(days=int(cfg.get("lookback_days", 3)))
    day = run_day.isoformat()
    http = Http(cfg.get("user_agent", "GameRadar/1.0"))
    sources = sources if sources is not None else build_sources(cfg, http, only)
    summary = {}
    for src in sources:
        src.skip = db.known_ids(src.name)
        fetched = added = 0
        start = run_day - timedelta(days=int(src.cfg.get("lookback_days", cfg.get("lookback_days", 3))))
        try:
            for g in src.fetch(start, run_day):
                fetched += 1
                rd = parse_date(g.release_date)
                if rd is None:
                    continue                      # undated: try again tomorrow
                if rd < start:
                    db.reject(g.source, g.source_id)  # old game: never fetch again
                    continue
                if rd > run_day:
                    continue                      # pre-release; picked up on launch day
                classify(g)
                added += db.upsert(g, day)
            db.log_run(day, src.name, fetched, added, "ok")
            log.info("%s: fetched %d, new %d", src.name, fetched, added)
        except SourceUnavailable as e:
            db.log_run(day, src.name, fetched, added, "skipped", str(e))
            log.warning("%s skipped: %s", src.name, e)
        except Exception as e:
            db.log_run(day, src.name, fetched, added, "error", repr(e))
            log.exception("%s failed after %d items (kept what was stored)", src.name, fetched)
        db.commit()
        summary[src.name] = added

    since = (run_day - timedelta(days=int(cfg.get("track_days", 30)))).isoformat()
    for src in sources:
        ids = db.recent_ids(src.name, since)
        n = 0
        try:
            for g in src.refresh(ids):
                db.snapshot(g, day)
                n += 1
        except SourceUnavailable:
            pass
        except Exception:
            log.exception("%s refresh failed", src.name)
        if n:
            log.info("%s: refreshed metrics for %d tracked games", src.name, n)
        db.commit()
    return summary

import csv
import html

from collections import Counter, defaultdict

from .db import norm_title

LINK_COLUMN_LABEL = {"steam_url": "Steam", "appstore_url": "App Store", "googleplay_url": "Google Play",
                     "itch_url": "itch.io", "epic_url": "Epic Games Store", "playstation_url": "PlayStation Store",
                     "xbox_url": "Xbox Store", "eshop_url": "Nintendo eShop", "website": "Website"}
LINK_COLUMNS = list(LINK_COLUMN_LABEL)
SOURCE_RANK = {"igdb": 0, "steam": 1, "appstore": 2, "googleplay": 3, "itch": 4}
STORE_LABEL = {"igdb": "IGDB", "steam": "Steam", "appstore": "App Store", "googleplay": "Google Play", "itch": "itch.io"}


def _merge_links(gs):
    links = {}
    for g in gs:
        own = g.links or ({STORE_LABEL.get(g.source, g.source): g.url} if g.url else {})
        for label, url in own.items():
            links.setdefault(label, url)
    return [{"label": k, "url": v} for k, v in ordered(links).items()]


def cluster(games):
    """Merge the same title seen on several stores into one row."""
    groups = defaultdict(list)
    for g in games:
        groups[norm_title(g.title) or g.key].append(g)
    rows = []
    for gs in groups.values():
        gs.sort(key=lambda g: SOURCE_RANK.get(g.source, 9))
        rep = gs[0]
        plats = sorted({p for g in gs for p in g.platforms})
        best = max(gs, key=lambda g: g.rating_count or 0)
        rows.append({
            "title": rep.title,
            "genre": min((g.genre for g in gs), key=lambda x: PRIORITY.get(x, 99)),
            "subgenres": sorted({s for g in gs for s in g.subgenres}),
            "platforms": plats,
            "families": sorted({platform_family(p) for p in plats}, key=FAMILIES.index),
            "release_date": min((g.release_date for g in gs if g.release_date), default=""),
            "developer": next((g.developer for g in gs if g.developer), ""),
            "publisher": next((g.publisher for g in gs if g.publisher), ""),
            "price": next((g.price for g in gs if g.price is not None), None),
            "currency": next((g.currency for g in gs if g.currency), ""),
            "free": any(g.is_free for g in gs),
            "iap": any(g.has_iap for g in gs),
            "rating": best.rating, "rating_count": best.rating_count,
            "installs": max((g.installs or 0 for g in gs), default=0) or None,
            "charts": sorted({c for g in gs for c in g.extra.get("charts", [])}),
            "appbrain": sorted({c for g in gs for c in g.extra.get("appbrain", [])}),
            "links": _merge_links(gs),
            "raw_genres": sorted({x for g in gs for x in g.raw_genres}),
            "tags": list(dict.fromkeys(t for g in gs for t in g.tags))[:12],
            "description": (rep.description or "")[:280],
            "icon": next((g.icon_url for g in gs if g.icon_url), ""),
        })
    rows.sort(key=lambda r: (PRIORITY.get(r["genre"], 99), -(r["rating_count"] or 0),
                             -len(r["charts"]), r["title"].lower()))
    return rows


def write_daily(db, day: date, out_root, basis="discovered", note=""):
    games = db.games_first_seen(day.isoformat()) if basis == "discovered" else db.games_released(day.isoformat())
    rows = cluster(games)
    out = Path(out_root) / day.isoformat()
    out.mkdir(parents=True, exist_ok=True)

    fields = ["source", "source_id", "title", "genre", "subgenres", "release_date", "platforms",
              "raw_genres", "tags", "game_modes", "developer", "publisher", "price", "currency",
              "is_free", "has_iap", "rating", "rating_count", "installs", "content_rating", "url"]
    with open(out / "games.csv", "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=fields + LINK_COLUMNS + ["all_links", "charts"])
        w.writeheader()
        for g in games:
            d = g.to_dict()
            row = {k: ("; ".join(map(str, d[k])) if isinstance(d[k], list) else d[k]) for k in fields}
            for col in LINK_COLUMNS:
                row[col] = g.links.get(LINK_COLUMN_LABEL[col], "")
            row["all_links"] = " | ".join(f"{k}: {v}" for k, v in ordered(g.links).items())
            row["charts"] = "; ".join(g.extra.get("charts", []))
            w.writerow(row)
    (out / "games.json").write_text(json.dumps(rows, indent=1, ensure_ascii=False), encoding="utf-8")
    (out / "summary.md").write_text(_markdown(day, rows, basis), encoding="utf-8")
    from .dashboard import aggregate, render, slim
    meta = {"updated": day.strftime("%d %b %Y").lstrip("0"), "csv": "games.csv", "repo": "", "note": note}
    (out / "report.html").write_text(render([slim(r) for r in rows], aggregate(rows), meta), encoding="utf-8")
    return out, rows


def _matrix(rows):
    m = defaultdict(Counter)
    for r in rows:
        for fam in r["families"] or ["Other"]:
            m[r["genre"]][fam] += 1
    return m


def _markdown(day, rows, basis):
    what = "newly discovered" if basis == "discovered" else "released"
    lines = [f"# New games, {day.isoformat()}", "",
             f"{len(rows)} unique games {what} (a game on several stores counts once).", ""]
    m = _matrix(rows)
    fams = [f for f in FAMILIES if any(m[g][f] for g in m)]
    by_genre = Counter(r["genre"] for r in rows)
    lines += ["| Genre | Total | " + " | ".join(fams) + " |", "|---|---:|" + "---:|" * len(fams)]
    for g in sorted(by_genre, key=lambda g: -by_genre[g]):
        lines.append(f"| {g} | {by_genre[g]} | " + " | ".join(str(m[g][f] or "") for f in fams) + " |")
    subs = Counter(s for r in rows for s in r["subgenres"])
    if subs:
        lines += ["", "**Sub-genres seen today:** " + ", ".join(f"{s} ({n})" for s, n in subs.most_common(15))]
    for g in GENRES:
        items = [r for r in rows if r["genre"] == g]
        if not items:
            continue
        lines += ["", f"## {g} ({len(items)})", ""]
        for r in items[:40]:
            bits = [", ".join(r["platforms"])]
            if r["developer"]:
                bits.append(r["developer"])
            if r["subgenres"]:
                bits.append("/".join(r["subgenres"]))
            if r["rating_count"]:
                bits.append(f"{r['rating']:.0f}/100 from {r['rating_count']:,} ratings" if r["rating"] else f"{r['rating_count']:,} ratings")
            if r["charts"]:
                bits.append(f"charting x{len(r['charts'])}")
            main = r["links"][0]["url"] if r["links"] else ""
            title = f"[{r['title']}]({main})" if main else r["title"]
            others = " · ".join(f"[{l['label']}]({l['url']})" for l in r["links"][1:])
            lines.append(f"- **{title}**: " + "; ".join(bits) + (f"  \n  {others}" if others else ""))
        if len(items) > 40:
            lines.append(f"- ...and {len(items) - 40} more in games.csv")
    return "\n".join(lines) + "\n"


def write_trends(db, end: date, days: int, out_root):
    start = end - timedelta(days=days - 1)
    prev_start = start - timedelta(days=days)
    cur = Counter()
    prev = Counter()
    daily = defaultdict(Counter)
    for r in db.genre_counts(prev_start.isoformat(), end.isoformat()):
        d = date.fromisoformat(r["day"])
        if d >= start:
            cur[r["genre"]] += r["n"]
            daily[r["day"]][r["genre"]] += r["n"]
        else:
            prev[r["genre"]] += r["n"]
    out = Path(out_root) / "trends"
    out.mkdir(parents=True, exist_ok=True)
    genres = [g for g in GENRES if cur[g] or prev[g]]
    with open(out / f"genre-by-day-{end.isoformat()}-{days}d.csv", "w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["release_date"] + genres)
        for d in sorted(daily):
            w.writerow([d] + [daily[d][g] for g in genres])
    total, ptotal = sum(cur.values()) or 1, sum(prev.values())
    lines = [f"# Genre trends: last {days} days to {end.isoformat()}", "",
             "| Genre | Releases | Share | Prior period | Change |", "|---|---:|---:|---:|---:|"]
    for g in sorted(genres, key=lambda g: -cur[g]):
        ch = f"{(cur[g] - prev[g]) / prev[g] * 100:+.0f}%" if prev[g] else "new"
        lines.append(f"| {g} | {cur[g]} | {cur[g] / total * 100:.1f}% | {prev[g]} | {ch if ptotal else 'n/a'} |")
    p = out / f"trends-{end.isoformat()}-{days}d.md"
    p.write_text("\n".join(lines) + "\n", encoding="utf-8")
    return p

def _genre_colors():
    # evenly spaced hues, alternating lightness so neighbours stay distinguishable
    return {g: f"hsl({(i * 137.5) % 360:.0f} {58 if i % 2 else 48}% {46 if i % 2 else 40}%)"
            for i, g in enumerate(GENRES)}


def _html(day, rows, basis, note=""):
    colors = _genre_colors()
    counts = Counter(r["genre"] for r in rows)
    data = json.dumps(rows, ensure_ascii=False).replace("</", "<\\/")
    meta = json.dumps({"colors": colors, "order": GENRES})
    what = "Discovered" if basis == "discovered" else "Released"
    return (HTML_TEMPLATE
            .replace("__DAY__", html.escape(day.strftime("%A %d %B %Y")))
            .replace("__COUNT__", str(len(rows)))
            .replace("__GENRES__", str(len(counts)))
            .replace("__NOTE__", html.escape(note) or "Click a genre band to show or hide it. A game on several stores is listed once.")
            .replace("__WHAT__", what)
            .replace("__DATA__", data).replace("__META__", meta))

"""The Game Radar dashboard: one accessible page that answers 'what launched, in what genres,
and what's getting traction' before showing the full list."""
import json
from collections import Counter, defaultdict

from .taxonomy import GENRES

# Mechanics a hybrid-casual mobile studio watches. Order = display order.
WATCH = ["Merge", "Match-3", "Sort & Organise", "Runner", "Hyper-casual", "Hybrid-casual", "Idle",
         "Squad / Idle RPG", "Survivors-like", "Tower Defense", "Auto-battler", "Gacha / Hero Collector",
         "Tycoon", ".io", "Roguelite", "Deckbuilder"]


CORE_STORES = {"Steam", "App Store", "Google Play", "Epic Games Store", "GOG", "PlayStation Store", "Xbox Store",
               "Microsoft Store", "Nintendo eShop", "Meta Quest Store", "IGDB"}


def itch_only(r):
    labels = {l["label"] for l in r.get("links") or r.get("l") or []}
    return not (labels & CORE_STORES)


def slim(r, seen=None):
    """Only the fields the page needs, to keep it light on phones."""
    return {
        "t": r["title"], "g": r["genre"], "s": r["subgenres"], "p": r["platforms"], "f": r["families"],
        "d": r["release_date"], "dev": r.get("developer") or "", "pr": r.get("price"),
        "cur": r.get("currency") or "", "free": bool(r.get("free")), "iap": bool(r.get("iap")),
        "ra": r.get("rating"), "rc": r.get("rating_count"), "in": r.get("installs"),
        "ch": r.get("charts") or [], "l": r.get("links") or [], "img": r.get("icon") or "",
        "desc": (r.get("description") or "")[:160], "seen": seen,
        "ab": r.get("appbrain") or [], "itch": itch_only(r),
    }


def aggregate(rows):
    """Per release-date counts of genres, sub-genres and platform families, both for all games
    ("all") and excluding itch.io-only uploads ("core")."""
    new = lambda: {"n": 0, "g": Counter(), "s": Counter(), "f": Counter()}
    agg = defaultdict(lambda: {"all": new(), "core": new()})
    for r in rows:
        d = r.get("release_date") or r.get("d")
        if not d:
            continue
        for bucket in (["all"] if itch_only(r) else ["all", "core"]):
            a = agg[d][bucket]
            a["n"] += 1
            a["g"][r.get("genre") or r.get("g")] += 1
            for s in r.get("subgenres") or r.get("s") or []:
                a["s"][s] += 1
            for f in r.get("families") or r.get("f") or []:
                a["f"][f] += 1
    plain = lambda v: {"n": v["n"], "g": dict(v["g"]), "s": dict(v["s"]), "f": dict(v["f"])}
    return {d: {k: plain(v) for k, v in b.items()} for d, b in sorted(agg.items())}


def render(rows, agg, meta):
    """rows: slim() dicts. agg: aggregate() output. meta: dict with updated, latest, csv, note, archive."""
    data = {"rows": rows, "agg": agg, "meta": meta, "genres": GENRES, "watch": WATCH}
    payload = json.dumps(data, ensure_ascii=False, separators=(",", ":")).replace("</", "<\\/")
    return TEMPLATE.replace("__DATA__", payload)

def test_labels_and_order():
    assert label_for("https://store.steampowered.com/app/1/")[0] == "Steam"
    assert label_for("https://maker.itch.io/game")[0] == "itch.io"
    assert label_for("https://www.nintendo.com/us/store/products/x")[0] == "Nintendo eShop"
    assert label_for("https://mygame.com")[0] == "Website"
    links = {"YouTube": "https://youtube.com/x", "Website": "https://g.com", "IGDB": "https://igdb.com/g", "Steam": "https://store.steampowered.com/app/1/"}
    assert list(ordered(links)) == ["Steam", "Website", "YouTube", "IGDB"]


def test_sources_collect_links():
    g = parse_game({"id": 1, "name": "X", "url": "https://www.igdb.com/games/x",
                    "websites": [{"url": "https://store.steampowered.com/app/9/"}, {"url": "https://store.playstation.com/x"},
                                 {"url": "https://x.com/game"}, {"url": "https://xgame.com"}]})
    assert {"Steam", "PlayStation Store", "Website", "X", "IGDB"} <= set(g.links)
    s = parse_details("9", {"type": "game", "name": "X", "website": "https://xgame.com", "release_date": {"date": "8 Oct, 2026"}})
    assert s.links == {"Steam": "https://store.steampowered.com/app/9/", "Website": "https://xgame.com"}
    a = parse_lookup({"kind": "software", "trackId": 5, "genres": ["Games"], "trackViewUrl": "https://apps.apple.com/app/id5",
                      "sellerUrl": "https://studio.com"}, "us")
    assert a.links == {"App Store": "https://apps.apple.com/app/id5", "Website": "https://studio.com"}


def test_merged_game_shows_every_store(tmp_path):
    db = DB(tmp_path / "r.db")
    day = "2026-10-08"
    for g in (Game(source="steam", source_id="9", title="Star Grid Arena", release_date=day, raw_genres=["Strategy"],
                   links={"Steam": "https://store.steampowered.com/app/9/", "Website": "https://sga.com"}),
              Game(source="googleplay", source_id="com.sga", title="Star Grid Arena", release_date=day, raw_genres=["Strategy"],
                   links={"Google Play": "https://play.google.com/store/apps/details?id=com.sga"})):
        db.upsert(classify(g), day)
    out, rows = write_daily(db, date(2026, 10, 8), tmp_path / "rep")
    assert [l["label"] for l in rows[0]["links"]] == ["Steam", "Google Play", "Website"]
    md = (out / "summary.md").read_text()
    assert "[Star Grid Arena](https://store.steampowered.com/app/9/)" in md and "[Google Play](https://play.google.com" in md
    csv_text = (out / "games.csv").read_text()
    assert "steam_url" in csv_text and "https://sga.com" in csv_text
    assert 'href="https://store.steampowered.com/app/9/"' not in (out / "report.html").read_text()  # data is JSON, rendered by JS
    assert "https://store.steampowered.com/app/9/" in (out / "report.html").read_text()


def test_steam_hover_tags_and_retag(tmp_path, monkeypatch):
    from radar.sources import steam
    html = '<div class="hover_tag_row"><div class="app_tag">Roguelite</div><div class="app_tag">Deckbuilding</div><div class="app_tag">Rock &amp; Roll</div></div>'
    assert steam.parse_hover_tags(html) == ["Roguelite", "Deckbuilding", "Rock & Roll"]
    from radar import pipeline
    db = DB(tmp_path / "r.db")
    g = classify(Game(source="steam", source_id="7", title="Cards of Doom", release_date="2026-10-01", raw_genres=["Action", "Indie"]))
    assert g.genre == "Action"
    db.upsert(g, "2026-10-02"); db.commit()
    monkeypatch.setattr(pipeline, "fetch_tags", lambda http, appid: ["Roguelite", "Deckbuilding"])
    assert pipeline.retag_steam({}, db) == ["2026-10-02"]
    row = db.c.execute("SELECT genre FROM games WHERE key='steam:7'").fetchone()
    assert row["genre"] == "Roguelike"
    assert pipeline.retag_steam({}, db) == []   # nothing left to do


def test_appbrain_parse_and_itch_flag():
    from radar.sources.googleplay import parse_appbrain
    html = ('<div class="sidebar"><a href="/app/x/com.side.bar">Side</a></div>'
            '<a class="browse-app safelink hover-shadow d-block" href="/app/overpeak-action/com.rusergames.overpeak"><h3>OverPeak</h3></a>'
            '<a href="/app/fruit-wins/com.fruitwins" class="browse-app safelink">Fruit</a>'
            '<a class="browse-app" href="/app/overpeak-action/com.rusergames.overpeak">dup</a>')
    assert parse_appbrain(html) == ["com.rusergames.overpeak", "com.fruitwins"]
    from radar.dashboard import aggregate, itch_only
    a = {"release_date": "2026-10-08", "genre": "Puzzle", "subgenres": [], "families": ["Web"], "links": [{"label": "itch.io", "url": "u"}]}
    b = {"release_date": "2026-10-08", "genre": "Puzzle", "subgenres": [], "families": ["Android"], "links": [{"label": "Google Play", "url": "u"}]}
    assert itch_only(a) and not itch_only(b)
    agg = aggregate([a, b])["2026-10-08"]
    assert agg["all"]["n"] == 2 and agg["core"]["n"] == 1
